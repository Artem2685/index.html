function calculateIsoscelesTriangle(base, height, offset) { // 1. Координаты исходного треугольника const x1 = 0, y1 = 0; const x2 = base, y2 = 0; const x3 = base / 2, y3 = height; // 2. Смещение сторон внутрь на offset (11 мм) // (Здесь используется упрощенная логика для примера, // в реальном приложении нужно реализовать полноценный алгоритм смещения линий) const newBase = base - 2 * offset * (base / (2 * height)); // Примерная формула const newSide = Math.sqrt(Math.pow(newBase / 2, 2) + Math.pow(height - offset, 2)); // 3. Возвращаем длины для отрисовки и вывода return { bottom: newBase.toFixed(1), left: newSide.toFixed(1), right: newSide.toFixed(1) }; } // Пример вызова const result = calculateIsoscelesTriangle(750, 637, 11); console.log("Низ:", result.bottom, "Бок:", result.left); // Выведет примерно: Низ: 737.0 Бок: 727.0
